---BEGIN FILE episodic_lab/arms.py---
"""Model arms of milestone 2 (protocol section 4): A0, A1 and the diagnostic O1.

Every answer is built from scratch: one fresh chat (system instruction + user message)
per question, no history, no cache across questions. The model is any object with
`generate(messages) -> str` and `weights_digest() -> str` (see models.py); tests use a
fake model, so the prompt logic is checked without PyTorch.

The three arms see exactly the same instruction (which names the public answer type of
the question) and question wording; they differ only in the story notes:
- A0 (no memory): no notes;
- A1 (all episodes in the prompt): every episode presented so far, in presentation order;
- O1 (oracle evidence, diagnostic only): the evaluator's target episodes of the question;
  for lures, confusable early episodes that each contain one pair of the lure's details
  (so the rejection needs binding, not just an empty context). O1 is not a memory
  system; it measures whether the model can use given evidence, its evidence is passed by
  the runner and never stored, and its E6 values say nothing about retention.
A0 is, strictly, a missing-evidence control with requested abstention.
"""
from __future__ import annotations

import hashlib
import json

# The answer type is public (it is part of every question record), so naming the
# expected form is fair; examples use names that never occur in any world.
FORMAT = {
    "person": ("a person's full name", "Tom Hill"),
    "place": ("a place name", "Elm Court"),
    "object": ("an object name", "red door"),
    "day": ("a day number", "Day 3"),
    "choice": ("the letter A or the letter B", "A"),
}
NO_NOTES = "(no notes)"


def instruction(answer_type: str) -> str:
    form, example = FORMAT[answer_type]
    text = (f"You read story notes and answer one question about them. Answer with only "
            f"{form} (for example: {example}), nothing else.")
    if answer_type != "choice":
        text += (" If the notes do not contain the event the question asks about, answer "
                 "with the single word: none.")
    return text


def build_messages(question: str, notes: list[str], answer_type: str) -> list[dict]:
    body = "\n".join(notes) if notes else NO_NOTES
    return [{"role": "system", "content": instruction(answer_type)},
            {"role": "user", "content": f"Story notes:\n{body}\n\nQuestion: {question}"}]


class _ModelArm:
    DIGEST_COVERS = ("model",)
    # E1b declaration per encoded episode (class attributes, not memory state)
    E1B = {"internal_replays": 0, "optimizer_steps": 0, "store_writes": 0}

    def __init__(self, model):
        self.model = model
        self.notes: list[str] = []

    def state_digest(self) -> str:
        return hashlib.sha256((self.model.weights_digest() + json.dumps(self.notes)).encode()).hexdigest()

    def deep_digest(self) -> str:
        """Recomputes the weights hash from the tensors (slow; checked once per phase)."""
        return self.model.weights_digest(recompute=True)


class NoMemory(_ModelArm):
    ARM = "A0"
    EXPOSURE = "none"  # E1b: no episode is ever placed in a prompt

    def encode(self, text: str) -> None:
        pass

    def answer(self, text: str, answer_type: str) -> str:
        return self.model.generate(build_messages(text, [], answer_type))


class AllInPrompt(_ModelArm):
    ARM = "A1"
    EXPOSURE = "all_presented"  # E1b: every presented episode is in every later prompt
    E1B = {"internal_replays": 0, "optimizer_steps": 0, "store_writes": 1}  # one note each

    def encode(self, text: str) -> None:
        self.notes.append(text)

    def answer(self, text: str, answer_type: str) -> str:
        return self.model.generate(build_messages(text, list(self.notes), answer_type))


class OracleEvidence(_ModelArm):
    ARM = "O1"
    EXPOSURE = "evidence"  # E1b: exactly the evidence passed with each question

    def encode(self, text: str) -> None:
        pass

    def answer(self, text: str, answer_type: str) -> str:  # pragma: no cover
        raise RuntimeError("O1 answers only with evidence")

    def answer_with(self, text: str, evidence: list[str], answer_type: str) -> str:
        return self.model.generate(build_messages(text, list(evidence), answer_type))


ARMS = {"A0": NoMemory, "A1": AllInPrompt, "O1": OracleEvidence}

---BEGIN FILE episodic_lab/runner.py---
"""Run one system on one world pair under the protocol's encoding and test schedule.

Schedule: present the early episodes once each in presentation order, test with the
phase1 questions, present the follow-up episodes once each, test with the final
questions. The system sees only episode texts and question texts (E1a); the evaluator
alone holds the gold answers.

Frozen memory during tests: every answer must come from a fresh context (the adapter
builds each prompt from scratch). The runner checks this as far as it can see: before
and after every answer it compares the system's own digest of its memory (for large
state such as weights, attributes listed in DIGEST_COVERS) AND a fingerprint of every
other attribute of the system object. A system that changes either while answering is
rejected. State kept outside the system object (module globals, files) is not covered;
model adapters must not keep any (protocol section 3, code audit). Systems with large
state (model weights) also offer `deep_digest()`, recomputed from the data itself at the
start and after every test phase.

Diagnostic oracle arms (O1) offer `answer_with(question, evidence)`; the runner passes
evaluator knowledge that the arm never stores: for positive items the texts of the
question's target episodes, for E5 lures confusable early episodes (for every pair of
the lure's cue details, the first early episode in presentation order that contains that
pair; deduplicated, in presentation order). A lure is therefore never answered from an
empty context; rejecting it needs the binding of the details.

E1b (processing per episode): every encoded episode is read once by `encode`. How often
an episode is placed in an answering prompt follows from the arm's declared EXPOSURE
("none", "all_presented", "evidence") and is counted here by the runner for every
question, so the count is part of the result and not of the arm's state.

`on_phase(phase, answers)` (optional) is called after each test phase with the raw
answers so far, so a caller can keep partial results if a later phase fails.
"""
from __future__ import annotations

import hashlib
import pickle
from typing import Protocol

from .questions import matching
from .score import score_world, summarise
from .world import episodes_of


class System(Protocol):
    def encode(self, text: str) -> None: ...
    def answer(self, text: str, answer_type: str) -> str: ...
    def state_digest(self) -> str: ...


class StateChangedWhileAnswering(RuntimeError):
    pass


def fingerprint(system) -> str:
    """Digest of the system's own memory digest plus every attribute it does not cover."""
    covered = set(getattr(system, "DIGEST_COVERS", ()))
    rest = {k: v for k, v in sorted(vars(system).items()) if k not in covered}
    return hashlib.sha256(system.state_digest().encode() + pickle.dumps(rest, protocol=4)).hexdigest()


def oracle_evidence(item: dict, episodes: list) -> list[str]:
    """Episode ids passed to O1: targets for positive items, confusable ones for lures."""
    order = {e.episode_id: e.position for e in episodes}
    if item["criterion"] != "E5":
        return sorted(item["targets"], key=order.get)
    early = sorted((e for e in episodes if e.block == "early"), key=lambda e: e.position)
    keys = sorted(item["cue"])
    chosen = []
    for i, a in enumerate(keys):
        for b in keys[i + 1:]:
            hits = matching(early, **{a: item["cue"][a], b: item["cue"][b]})
            if not hits:
                raise ValueError(f"lure pair {a},{b} has no early episode")
            if hits[0].episode_id not in chosen:
                chosen.append(hits[0].episode_id)
    return sorted(chosen, key=order.get)


EXPOSURES = ("none", "all_presented", "evidence")


def run_world(system: System, world: dict, questions: dict, on_phase=None) -> dict:
    episodes = episodes_of(world)
    texts = {e.episode_id: e.text for e in episodes}
    oracle = hasattr(system, "answer_with")
    exposure = getattr(system, "EXPOSURE", None)
    if exposure is not None and exposure not in EXPOSURES:
        raise ValueError(f"unknown EXPOSURE {exposure!r}")
    if oracle and exposure != "evidence":
        raise ValueError("oracle arms must declare EXPOSURE = 'evidence'")
    deep = getattr(system, "deep_digest", None)
    deep_start = deep() if deep else None
    presented = []
    in_prompt = {e.episode_id: 0 for e in episodes}
    encoded = {e.episode_id: 0 for e in episodes}
    asked = []
    answers = {}
    for block, phase in (("early", "phase1"), ("followup", "final")):
        for e in episodes:
            if e.block == block:
                system.encode(e.text)
                presented.append(e.episode_id)
                encoded[e.episode_id] += 1
        answers[phase] = {}
        for q in questions["public"][phase]:
            before = fingerprint(system)
            if oracle:
                evidence = oracle_evidence(questions["gold"][q["qid"]], episodes)
                reply = system.answer_with(q["text"], [texts[t] for t in evidence],
                                           q["answer_type"])
                shown = evidence
            else:
                reply = system.answer(q["text"], q["answer_type"])
                shown = presented if exposure == "all_presented" else []
            for t in shown:
                in_prompt[t] += 1
            answers[phase][q["qid"]] = reply
            asked.append([phase, q["qid"]])
            if fingerprint(system) != before:
                raise StateChangedWhileAnswering(q["qid"])
        if deep and deep() != deep_start:
            raise StateChangedWhileAnswering(f"weights changed during {phase}")
        if on_phase:
            on_phase(phase, dict(answers[phase]))
    gold, cands = questions["gold"], questions["candidates"]
    rows = (score_world(gold, cands, answers["phase1"], "phase1")
            + score_world(gold, cands, answers["final"], "final"))
    return {"seed": world["seed"], "overlap": world["overlap"],
            "episodes_presented": len(presented), "oracle": oracle,
            "episodes_in_world": len(episodes), "rows": rows, "asked": asked,
            "processing": {"exposure": exposure, "encode_reads": encoded,
                           "declared_per_episode": getattr(system, "E1B", None),
                           "prompt_inclusions": in_prompt if exposure else None},
            "phase1": summarise([r for r in rows if r["phase"] == "phase1"]),
            "final": summarise([r for r in rows if r["phase"] == "final"])}


def retention(result: dict) -> dict:
    """E6 inputs: accuracy on the early-target items at phase1 and at final."""
    early = [r for r in result["rows"] if r["criterion"] in ("E2", "E4")]
    out = {}
    for phase in ("phase1", "final"):
        sub = [r for r in early if r["phase"] == phase]
        out[phase] = sum(r["correct"] for r in sub) / len(sub) if sub else float("nan")
    return out

---BEGIN FILE bench/pilot/run_pilot.py---
"""Pilot cell: one model, one arm, one world (both overlap conditions).

Writes a JSON result with every raw answer, the scored rows (each with its prompt length,
answer length and a truncation flag), summaries, E6 inputs, timing and token telemetry,
E1b processing counts per episode, provenance (code commit, SHA-256 of the pilot request
and of the pins manifest, SHA-256 of every world and question set), plus whether the
symbolic arm solves the same world (reference). On any failure the result keeps what was
done (finished conditions, raw answers of finished test phases under "partial") and
status ERROR; the exit code is then non-zero.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
import sys
import time
import traceback
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))

from episodic_lab.arms import ARMS  # noqa: E402
from episodic_lab.questions import make_world  # noqa: E402
from episodic_lab.runner import retention, run_world  # noqa: E402
from episodic_lab.symbolic import SymbolicMemory  # noqa: E402

SCHEMA = "episodic_lab.pilot_cell.v2"


def canonical_sha256(obj) -> str:
    return hashlib.sha256(json.dumps(obj, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def file_sha256(path: str) -> str:
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def inclusion_summary(counts: dict) -> dict:
    values = list(counts.values())
    return {"episodes": len(values), "total": sum(values), "min": min(values),
            "max": max(values), "mean": sum(values) / len(values)}


def run_cell(model, arm: str, series: str, index: int, out: dict) -> None:
    attempt, pair = make_world(series, index)
    out.update({"series": series, "index": index, "attempt": attempt, "conditions": {},
                "partial": {}})
    for overlap, (world, questions) in pair.items():
        t0 = time.perf_counter()
        before = dict(model.stats)
        first_call = len(model.call_log)
        out["partial"][overlap] = {}
        result = run_world(ARMS[arm](model), world, questions,
                           on_phase=lambda phase, answers: out["partial"][overlap].update({phase: answers}))
        calls = model.call_log[first_call:]
        if len(calls) != len(result["asked"]):
            raise RuntimeError(f"{len(calls)} model calls for {len(result['asked'])} questions")
        per_answer = {tuple(k): c for k, c in zip(map(tuple, result["asked"]), calls)}
        for row in result["rows"]:
            row.update(per_answer[(row["phase"], row["qid"])])
        reference = run_world(SymbolicMemory(), world, questions)
        processing = result["processing"]
        tokens = {e["episode_id"]: model.count_tokens(e["text"]) for e in world["episodes"]}
        processing["episode_tokens"] = tokens
        processing["prompt_tokens_per_episode"] = {
            k: n * tokens[k] for k, n in processing["prompt_inclusions"].items()}
        out["conditions"][overlap] = {
            "seed": world["seed"], "seconds": round(time.perf_counter() - t0, 1),
            "world_sha256": canonical_sha256(world),
            "questions_sha256": canonical_sha256(questions),
            "telemetry": {k: model.stats[k] - before[k] for k in model.stats},
            "phase1": result["phase1"], "final": result["final"],
            "retention": retention(result), "rows": result["rows"],
            "episodes_presented": result["episodes_presented"],
            "processing": dict(processing,
                               prompt_inclusion_summary=inclusion_summary(processing["prompt_inclusions"])),
            "symbolic_all_correct": all(r["correct"] for r in reference["rows"])}
        del out["partial"][overlap]
    del out["partial"]


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--model", required=True)
    parser.add_argument("--arm", required=True, choices=sorted(ARMS))
    parser.add_argument("--series", required=True)
    parser.add_argument("--index", type=int, required=True)
    parser.add_argument("--pins", default="config/model_pins.json")
    parser.add_argument("--request", default="config/pilot.json")
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    out = {"schema": SCHEMA, "model": args.model, "arm": args.arm, "series": args.series,
           "index": args.index, "status": "ERROR",
           "provenance": {"code_commit": os.environ.get("GITHUB_SHA", "local"),
                          "request_sha256": file_sha256(args.request),
                          "pins_sha256": file_sha256(args.pins)}}
    t0 = time.perf_counter()
    try:
        from episodic_lab.models import PinnedModel
        model = PinnedModel(args.model, args.pins)
        out.update({"repo": model.repo, "revision": model.revision,
                    "max_new_tokens": model.max_new_tokens,
                    "weights_digest": model.weights_digest()})
        run_cell(model, args.arm, args.series, args.index, out)
        out["status"] = "OK"
    except Exception:
        out["traceback"] = traceback.format_exc()[-4000:]
    out["total_seconds"] = round(time.perf_counter() - t0, 1)
    Path(args.out).write_text(json.dumps(out, indent=1, sort_keys=True) + "\n")
    print(json.dumps({k: v for k, v in out.items() if k not in ("conditions", "partial")},
                     sort_keys=True))
    return 0 if out["status"] == "OK" else 1


if __name__ == "__main__":
    raise SystemExit(main())
